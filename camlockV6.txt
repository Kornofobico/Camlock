-- Helper para checar disponibilidade de API de arquivos do executor
local function fileExists(path)
    if type(isfile) == "function" then
        local ok, res = pcall(isfile, path)
        if ok then return res end
    end
    if type(readfile) == "function" then
        local ok, _ = pcall(readfile, path)
        return ok
    end
    return false
end

local function canWriteFile()
    return type(writefile) == "function"
end

local function saveConfig()
    if not canWriteFile() then return end
    local cfg = {
        aimlockEnabled = aimlockEnabled,
        autoFireEnabled = autoFireEnabled,
        maxDistance = maxDistance,
        lerpSpeed = lerpSpeed,
        fov = fov,
        selectedPart = selectedPart,
        whitelistedTeams = whitelistedTeams,
        floatingPosition = (floatingToggleButton and floatingToggleButton.Position and {
            X = floatingToggleButton.Position.X.Scale,
            XOff = floatingToggleButton.Position.X.Offset,
            Y = floatingToggleButton.Position.Y.Scale,
            YOff = floatingToggleButton.Position.Y.Offset
        }) or nil,
        aimOffset = { x = aimOffset.X, y = aimOffset.Y, z = aimOffset.Z }
    }
    local ok, err = pcall(function()
        writefile(configPath, HttpService:JSONEncode(cfg))
    end)
    if not ok then
        -- falhar silenciosamente se executor não permitir
        -- print("Failed to save Camlock config:", err)
    end
end

local function loadConfig()
    if type(readfile) ~= "function" then return end
    if not fileExists(configPath) then return end
    local content
    local ok, err = pcall(function()
        content = readfile(configPath)
    end)
    if not ok or not content then return end
    local suc, data = pcall(function() return HttpService:JSONDecode(content) end)
    if not suc or type(data) ~= "table" then return end

    aimlockEnabled = (data.aimlockEnabled == true) or aimlockEnabled
    autoFireEnabled = (data.autoFireEnabled == true) or autoFireEnabled
    maxDistance = (type(data.maxDistance) == "number") and data.maxDistance or maxDistance
    lerpSpeed = (type(data.lerpSpeed) == "number") and data.lerpSpeed or lerpSpeed
    fov = (type(data.fov) == "number") and data.fov or fov
    selectedPart = (type(data.selectedPart) == "string") and data.selectedPart or selectedPart
    if type(data.whitelistedTeams) == "table" then
        whitelistedTeams = data.whitelistedTeams
    end

    if type(data.aimOffset) == "table" then
        local ax = tonumber(data.aimOffset.x) or 0
        local ay = tonumber(data.aimOffset.y) or 0
        local az = tonumber(data.aimOffset.z) or 0
        aimOffset = Vector3.new(ax, ay, az)
    end

    return data.floatingPosition
end

--// Util
local function onScreenAndVisible(part: BasePart)
    if not part then return false end
    local targetPos = part.Position + aimOffset

    local _, onScreen = Camera:WorldToViewportPoint(targetPos)
    if not onScreen then return false end

    local origin = Camera.CFrame.Position
    local direction = (targetPos - origin)
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    local result = workspace:Raycast(origin, direction, rayParams)
    return not result or (result.Instance and result.Instance:IsDescendantOf(part.Parent))
end

local function isTargetValid(targetPart: BasePart)
    if not targetPart or not targetPart.Parent then return false end
    local char = targetPart.Parent
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local targetPos = targetPart.Position + aimOffset
    if (targetPos - Camera.CFrame.Position).Magnitude > maxDistance then return false end
    if not onScreenAndVisible(targetPart) then return false end
    return true
end

local function getTeamName(player: Player)
    return (player.Team and player.Team.Name) or "__NO_TEAM__"
end

local function isInFOV(targetPart: BasePart)
    local targetPos = targetPart.Position + aimOffset
    local screenPoint, onScreen = Camera:WorldToViewportPoint(targetPos)
    if not onScreen then return false end

    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local point = Vector2.new(screenPoint.X, screenPoint.Y)
    local distance = (point - center).Magnitude

    return distance <= fov
end

local function getNearestEnemyPart()
    local closest: BasePart? = nil
    local shortest = math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local tname = getTeamName(p)
            if not whitelistedTeams[tname] then
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                local part = p.Character:FindFirstChild(selectedPart)
                if hum and hum.Health > 0 and part and part:IsA("BasePart") then
                    local dist = (part.Position + aimOffset - Camera.CFrame.Position).Magnitude
                    if dist < maxDistance and onScreenAndVisible(part) and dist < shortest and isInFOV(part) then
                        shortest = dist
                        closest = part
                    end
                end
            end
        end
    end
    return closest
end

--// Auto Fire
local function performAutoFire()
    if not autoFireEnabled then return end
    if not currentTarget or not isTargetValid(currentTarget) then return end

    local currentTime = tick()
    if currentTime - lastAutoFireTime >= autoFireDelay then
        lastAutoFireTime = currentTime

        pcall(function()
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.ButtonR2, false, nil)
            task.wait(0.05)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.ButtonR2, false, nil)
        end)
    end
end

local function enableDrag(handle: Instance, dragTarget: Instance)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = dragTarget.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            dragTarget.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

    floatingToggle.TextColor3 = Color3.new(1,1,1)
    floatingToggle.Font = Enum.Font.GothamBold
    floatingToggle.TextSize = 14
    floatingToggle.Parent = screenGui
    Instance.new("UICorner", floatingToggle).CornerRadius = UDim.new(0, 8)
    floatingToggleButton = floatingToggle

    floatingToggle.MouseButton1Click:Connect(function()
        setAimlockState(not aimlockEnabled)
    end)

    local function draggable(btn)
        local dragging = false
        local dragStart = nil
        local startPos = nil

        btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = btn.Position

                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                        pcall(saveConfig)
                    end
                end)
            end
        end)

        btn.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart
                btn.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + delta.X,
                    startPos.Y.Scale, startPos.Y.Offset + delta.Y
                )
            end
        end)
    end
    draggable(floatingToggle)

    setAimlockState(aimlockEnabled)
end
